server {
    listen 80 default_server;
    listen [::]:80 default_server;

    server_name _;
    return 404;
}

server {
    listen 80;
    listen [::]:80;

    root /var/www/dist;
    index index.html;

    server_name analytics-uvao.ru www.analytics-uvao.ru;

    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    listen [::]:443 ssl ipv6only=on;

    root /var/www/dist;
    index index.html;

    server_name analytics-uvao.ru www.analytics-uvao.ru;

    ssl_certificate /etc/letsencrypt/live/analytics-uvao.ru/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/analytics-uvao.ru/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    # ⭐ ОТЧЁТЫ (проверь, что этот блок есть!)
    location /reports/ {
        alias /var/www/reports/;
        autoindex off;
        add_header Content-Disposition "attachment";
        add_header Cache-Control "no-store, no-cache, must-revalidate";
        add_header Pragma "no-cache";
        add_header Expires "0";
    }

    # React приложение
    location / {
        try_files $uri $uri/ /index.html;
        add_header Cache-Control "no-store, no-cache, must-revalidate";
        add_header Pragma "no-cache";
        add_header Expires "0";
    }

    # Flask API
    location /api {
        proxy_pass http://127.0.0.1:5000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
    }

    # Статика
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|json)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
        try_files $uri =404;
    }

    # Index.html
    location = /index.html {
        add_header Cache-Control "no-store, no-cache, must-revalidate";
        add_header Pragma "no-cache";
        add_header Expires "0";
    }
}
